#                                          last_growth_us
#                                        + SAFETY_RING_RECOVER_MS = 5000
#   cpcu_v2/src/cpcu_safety.c (v2.3)   — Ring fault threshold applied to
#                                        delta-since-baseline; clears on
#                                        5 s quiescence. API unchanged.
#   cpcu_v2/src/cpcu_tui.c (v3.4)      — Split into three .c files plus the
#   cpcu_v2/src/cpcu_tui_render.c        new include/cpcu_tui.h header.

#   cpcu_v2/include/cpcu_tui.h           from page 5 to page 7; live data
#                                        pages now occupy keys 1-6. IO
#                                        heartbeat thresholds expressed
#                                        relative to the 100 ms heartbeat
